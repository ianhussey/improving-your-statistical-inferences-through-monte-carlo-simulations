data <- dat_depression_rct


calc_summary <- function(data, grouping, variable){
  output <- data %>%
    group_by({{ grouping }}) %>%
    summarize(mean = mean({{ variable }}),
              sd = sd({{ variable }}),
              n = n())
  return(output)
}

                 n_items = 13,
